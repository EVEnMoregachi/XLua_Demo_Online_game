NetGetTavernInIt = {}
local net = require("Net/Base/NetGetTavernInItBase")
net.__index = net

function NetGetTavernInIt.Create(callback)
    local t = {}
	setmetatable(t, net)
    t:Init(callback)
    return t
end 

function net:OnResult(v)
    self.datav = v.value.data    
    tavern.PTheroData = self.datav.commonDrinkHeroList
    tavern.HDheroData = self.datav.activeDrinkHeroList 
    tavern.NowDrinkHero = self.datav.tavernInIt.nowDrinkHero
    tavern.DrinkActiveTime = self.datav.tavernInIt.drinkActiveTime
    tavern.ShaodaoFreeNum = self.datav.tavernInIt.shaodaoFreeNum
    tavern.NverhongFreeTime = self.datav.tavernInIt.nverhongFreeTime
    tavern.BaicaoFreeTime = self.datav.tavernInIt.baicaoFreeTime 
    tavern.NowDrinkHeroId =  ""
    tavern.refNum =self.datav.refNum
     tavern.IsGet = self.datav.succeId

end
 
 --获取普通弟子信息
function NetGetTavernInIt.getHeroPTMsg(heroId)  
    for i,v1 in ipairs(tavern.PTheroData) do  
          if v1.heroId == heroId then
           return v1 
          end
    end 
    return nil
end

 --获取活动弟子信息
 function NetGetTavernInIt.getHeroHDMsg(heroId)  
    for i,v1 in ipairs(tavern.HDheroData) do  
          if v1.heroId == heroId then
           return v1 
          end
    end 
    return nil
end
