NetHeroDirection = {}
local net = require("Net/Base/NetHeroDirectionBase")
net.__index = net

function NetHeroDirection.Create(heroId,pointType,callback)
    local t = {}
	setmetatable(t, net)
    t:Init(heroId,pointType,callback)
    return t
end 

function net:OnResult(v)
    -- print("VVVVVVVVVVVVVVVVVVVV")
    self.datav = v.value.data
end
