local NetGetSunBattleInitBase = {}

--阳七星初始化
function NetGetSunBattleInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetSunBattleInit"

	self.callback = callback

    return self
end

function NetGetSunBattleInitBase:Send()
    Net.Send(self)
end

function NetGetSunBattleInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetSunBattleInitBase
