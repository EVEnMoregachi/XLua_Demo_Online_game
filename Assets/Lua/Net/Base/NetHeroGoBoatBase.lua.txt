local NetHeroGoBoatBase = {}

--弟子行舟
-- heroId(string): 行舟弟子id
function NetHeroGoBoatBase:Init(heroId,callback)
    self.data = {}
	self.data.msgName = "HeroGoBoat"

	self.data.heroId = heroId
	self.callback = callback

    return self
end

function NetHeroGoBoatBase:Send()
    Net.Send(self)
end

function NetHeroGoBoatBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetHeroGoBoatBase
