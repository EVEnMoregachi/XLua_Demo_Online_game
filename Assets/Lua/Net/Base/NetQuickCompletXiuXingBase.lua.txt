local NetQuickCompletXiuXingBase = {}

--快速完成修行
-- xiuXingType(string): 修行的10种类型
function NetQuickCompletXiuXingBase:Init(xiuXingType,callback)
    self.data = {}
	self.data.msgName = "QuickCompletXiuXing"

	self.data.xiuXingType = xiuXingType
	self.callback = callback

    return self
end

function NetQuickCompletXiuXingBase:Send()
    Net.Send(self)
end

function NetQuickCompletXiuXingBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetQuickCompletXiuXingBase
