local NetInitEatChickBase = {}

--吃鸡界面初始化
function NetInitEatChickBase:Init(callback)
    self.data = {}
	self.data.msgName = "InitEatChick"

	self.callback = callback

    return self
end

function NetInitEatChickBase:Send()
    Net.Send(self)
end

function NetInitEatChickBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetInitEatChickBase
