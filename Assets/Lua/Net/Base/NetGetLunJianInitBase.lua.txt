local NetGetLunJianInitBase = {}

--论剑初始化
function NetGetLunJianInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetLunJianInit"

	self.callback = callback

    return self
end

function NetGetLunJianInitBase:Send()
    Net.Send(self)
end

function NetGetLunJianInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetLunJianInitBase
