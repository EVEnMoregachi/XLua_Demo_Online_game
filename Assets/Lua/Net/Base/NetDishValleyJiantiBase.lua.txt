local NetDishValleyJiantiBase = {}

--蝶谷医仙健体
-- heroId(string): 弟子id
function NetDishValleyJiantiBase:Init(heroId,callback)
    self.data = {}
	self.data.msgName = "DishValleyJianti"

	self.data.heroId = heroId
	self.callback = callback

    return self
end

function NetDishValleyJiantiBase:Send()
    Net.Send(self)
end

function NetDishValleyJiantiBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetDishValleyJiantiBase
