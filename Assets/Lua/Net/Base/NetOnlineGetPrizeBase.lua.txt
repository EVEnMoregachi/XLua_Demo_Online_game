local NetOnlineGetPrizeBase = {}

--在线奖励领取
-- prize_id(string): 奖励id
function NetOnlineGetPrizeBase:Init(prize_id,callback)
    self.data = {}
	self.data.msgName = "OnlineGetPrize"

	self.data.prize_id = prize_id
	self.callback = callback

    return self
end

function NetOnlineGetPrizeBase:Send()
    Net.Send(self)
end

function NetOnlineGetPrizeBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetOnlineGetPrizeBase
