local NetHeroVeinSysBase = {}

--经脉系统
-- heroId(string): 弟子id
function NetHeroVeinSysBase:Init(heroId,callback)
    self.data = {}
	self.data.msgName = "HeroVeinSys"

	self.data.heroId = heroId
	self.callback = callback

    return self
end

function NetHeroVeinSysBase:Send()
    Net.Send(self)
end

function NetHeroVeinSysBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetHeroVeinSysBase
