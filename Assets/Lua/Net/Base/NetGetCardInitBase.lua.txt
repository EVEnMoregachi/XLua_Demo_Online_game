local NetGetCardInitBase = {}

--抽卡初始化
function NetGetCardInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetCardInit"

	self.callback = callback

    return self
end

function NetGetCardInitBase:Send()
    Net.Send(self)
end

function NetGetCardInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetCardInitBase
