local NetGetOpenQiyuPageBase = {}

--拿到奇遇开放的功能
function NetGetOpenQiyuPageBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetOpenQiyuPage"

	self.callback = callback

    return self
end

function NetGetOpenQiyuPageBase:Send()
    Net.Send(self)
end

function NetGetOpenQiyuPageBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetOpenQiyuPageBase
