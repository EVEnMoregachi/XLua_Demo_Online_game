local NetGetPackageInitBase = {}

--初始化背包
function NetGetPackageInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetPackageInit"

	self.callback = callback

    return self
end

function NetGetPackageInitBase:Send()
    Net.Send(self)
end

function NetGetPackageInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetPackageInitBase
