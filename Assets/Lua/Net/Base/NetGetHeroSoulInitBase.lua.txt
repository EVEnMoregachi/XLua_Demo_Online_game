local NetGetHeroSoulInitBase = {}

--弟子魂魄初始化
function NetGetHeroSoulInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetHeroSoulInit"

	self.callback = callback

    return self
end

function NetGetHeroSoulInitBase:Send()
    Net.Send(self)
end

function NetGetHeroSoulInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetHeroSoulInitBase
