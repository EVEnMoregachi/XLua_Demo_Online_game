local NetGetRetreatInItBase = {}

--闭关初始化
function NetGetRetreatInItBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetRetreatInIt"

	self.callback = callback

    return self
end

function NetGetRetreatInItBase:Send()
    Net.Send(self)
end

function NetGetRetreatInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetRetreatInItBase
