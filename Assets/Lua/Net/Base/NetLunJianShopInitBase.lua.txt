local NetLunJianShopInitBase = {}

--论剑商店初始化
function NetLunJianShopInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "LunJianShopInit"

	self.callback = callback

    return self
end

function NetLunJianShopInitBase:Send()
    Net.Send(self)
end

function NetLunJianShopInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetLunJianShopInitBase
