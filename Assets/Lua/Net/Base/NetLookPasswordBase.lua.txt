local NetLookPasswordBase = {}

--道具密函查看密码
function NetLookPasswordBase:Init(callback)
    self.data = {}
	self.data.msgName = "LookPassword"

	self.callback = callback

    return self
end

function NetLookPasswordBase:Send()
    Net.Send(self)
end

function NetLookPasswordBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetLookPasswordBase
