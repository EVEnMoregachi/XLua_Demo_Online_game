local NetMentOneGetPrizeBase = {}

--成就一键领取
function NetMentOneGetPrizeBase:Init(callback)
    self.data = {}
	self.data.msgName = "MentOneGetPrize"

	self.callback = callback

    return self
end

function NetMentOneGetPrizeBase:Send()
    Net.Send(self)
end

function NetMentOneGetPrizeBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetMentOneGetPrizeBase
