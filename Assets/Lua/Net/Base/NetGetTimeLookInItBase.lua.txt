local NetGetTimeLookInItBase = {}

--限时寻侠初始化
function NetGetTimeLookInItBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetTimeLookInIt"

	self.callback = callback

    return self
end

function NetGetTimeLookInItBase:Send()
    Net.Send(self)
end

function NetGetTimeLookInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetTimeLookInItBase
