local NetGetTimeLookRankingBase = {}

--限时寻侠排行榜
function NetGetTimeLookRankingBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetTimeLookRanking"

	self.callback = callback

    return self
end

function NetGetTimeLookRankingBase:Send()
    Net.Send(self)
end

function NetGetTimeLookRankingBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetTimeLookRankingBase
