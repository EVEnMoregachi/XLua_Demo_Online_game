local NetGetBattleArrayInitBase = {}

--阵容初始化
function NetGetBattleArrayInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetBattleArrayInit"

	self.callback = callback

    return self
end

function NetGetBattleArrayInitBase:Send()
    Net.Send(self)
end

function NetGetBattleArrayInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetBattleArrayInitBase
