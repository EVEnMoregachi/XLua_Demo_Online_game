local NetSunBattleUpdateBase = {}

--阳七星阵容更新
-- heroId(string): 弟子id
-- posid(int): 上阵位置id
function NetSunBattleUpdateBase:Init(heroId,posid,callback)
    self.data = {}
	self.data.msgName = "SunBattleUpdate"

	self.data.heroId = heroId
	self.data.posid = posid
	self.callback = callback

    return self
end

function NetSunBattleUpdateBase:Send()
    Net.Send(self)
end

function NetSunBattleUpdateBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetSunBattleUpdateBase
