local NetXiaKeDaoInitBase = {}

--侠客岛初始化
function NetXiaKeDaoInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "XiaKeDaoInit"

	self.callback = callback

    return self
end

function NetXiaKeDaoInitBase:Send()
    Net.Send(self)
end

function NetXiaKeDaoInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetXiaKeDaoInitBase
