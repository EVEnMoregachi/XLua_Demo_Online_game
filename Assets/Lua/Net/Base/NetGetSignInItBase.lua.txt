local NetGetSignInItBase = {}

--签到初始化
function NetGetSignInItBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetSignInIt"

	self.callback = callback

    return self
end

function NetGetSignInItBase:Send()
    Net.Send(self)
end

function NetGetSignInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetSignInItBase
