local NetTavernRefreshBase = {}

--酒馆刷新普通弟子列表
function NetTavernRefreshBase:Init(callback)
    self.data = {}
	self.data.msgName = "TavernRefresh"

	self.callback = callback

    return self
end

function NetTavernRefreshBase:Send()
    Net.Send(self)
end

function NetTavernRefreshBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetTavernRefreshBase
