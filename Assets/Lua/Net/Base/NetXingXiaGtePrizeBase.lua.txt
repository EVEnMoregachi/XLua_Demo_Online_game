local NetXingXiaGtePrizeBase = {}

--行侠领奖
function NetXingXiaGtePrizeBase:Init(callback)
    self.data = {}
	self.data.msgName = "XingXiaGtePrize"

	self.callback = callback

    return self
end

function NetXingXiaGtePrizeBase:Send()
    Net.Send(self)
end

function NetXingXiaGtePrizeBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetXingXiaGtePrizeBase
