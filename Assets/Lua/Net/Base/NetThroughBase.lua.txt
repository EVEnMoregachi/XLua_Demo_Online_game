local NetThroughBase = {}

--出关
-- heroId(string): 行舟弟子id
function NetThroughBase:Init(heroId,callback)
    self.data = {}
	self.data.msgName = "Through"

	self.data.heroId = heroId
	self.callback = callback

    return self
end

function NetThroughBase:Send()
    Net.Send(self)
end

function NetThroughBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetThroughBase
