local NetGetMapTaskInItBase = {}

--初始化当前任务
-- map_id(string): 地图id
function NetGetMapTaskInItBase:Init(map_id,callback)
    self.data = {}
	self.data.msgName = "GetMapTaskInIt"

	self.data.map_id = map_id
	self.callback = callback

    return self
end

function NetGetMapTaskInItBase:Send()
    Net.Send(self)
end

function NetGetMapTaskInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetMapTaskInItBase
