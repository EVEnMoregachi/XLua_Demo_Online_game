local NetSendBulletScreenBase = {}

--BOSS战发送弹幕
-- chat_info(string): 聊天内容
-- boss_id(string): BOSSID
function NetSendBulletScreenBase:Init(chat_info,boss_id,callback)
    self.data = {}
	self.data.msgName = "SendBulletScreen"

	self.data.chat_info = chat_info
	self.data.boss_id = boss_id
	self.callback = callback

    return self
end

function NetSendBulletScreenBase:Send()
    Net.Send(self)
end

function NetSendBulletScreenBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetSendBulletScreenBase
