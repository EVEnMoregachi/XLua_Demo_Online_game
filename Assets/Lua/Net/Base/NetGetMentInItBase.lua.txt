local NetGetMentInItBase = {}

--获取成就初始化
function NetGetMentInItBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetMentInIt"

	self.callback = callback

    return self
end

function NetGetMentInItBase:Send()
    Net.Send(self)
end

function NetGetMentInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetMentInItBase
