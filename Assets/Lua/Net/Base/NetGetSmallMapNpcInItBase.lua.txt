local NetGetSmallMapNpcInItBase = {}

--初始化小地图NPC
-- map_id(string): 地图id
function NetGetSmallMapNpcInItBase:Init(map_id,callback)
    self.data = {}
	self.data.msgName = "GetSmallMapNpcInIt"

	self.data.map_id = map_id
	self.callback = callback

    return self
end

function NetGetSmallMapNpcInItBase:Send()
    Net.Send(self)
end

function NetGetSmallMapNpcInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetSmallMapNpcInItBase
