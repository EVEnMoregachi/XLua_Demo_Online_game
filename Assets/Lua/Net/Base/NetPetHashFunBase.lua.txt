local NetPetHashFunBase = {}

--宠物散功
-- petId(string): 宠物id
-- hashType(int): 散功类型（0 普通散工 1九转散功）
function NetPetHashFunBase:Init(petId,hashType,callback)
    self.data = {}
	self.data.msgName = "PetHashFun"

	self.data.petId = petId
	self.data.hashType = hashType
	self.callback = callback

    return self
end

function NetPetHashFunBase:Send()
    Net.Send(self)
end

function NetPetHashFunBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetPetHashFunBase
