local NetBattleArrayUpGoBase = {}

--阵容上阵
-- heroId(string): 弟子id
-- posid(int): 上阵位置id
function NetBattleArrayUpGoBase:Init(heroId,posid,callback)
    self.data = {}
	self.data.msgName = "BattleArrayUpGo"

	self.data.heroId = heroId
	self.data.posid = posid
	self.callback = callback

    return self
end

function NetBattleArrayUpGoBase:Send()
    Net.Send(self)
end

function NetBattleArrayUpGoBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetBattleArrayUpGoBase
