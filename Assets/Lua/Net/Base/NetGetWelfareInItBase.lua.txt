local NetGetWelfareInItBase = {}

--获取福利初始化
function NetGetWelfareInItBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetWelfareInIt"

	self.callback = callback

    return self
end

function NetGetWelfareInItBase:Send()
    Net.Send(self)
end

function NetGetWelfareInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetWelfareInItBase
