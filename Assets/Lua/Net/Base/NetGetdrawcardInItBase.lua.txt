local NetGetdrawcardInItBase = {}

--召唤抽卡初始化
function NetGetdrawcardInItBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetdrawcardInIt"

	self.callback = callback

    return self
end

function NetGetdrawcardInItBase:Send()
    Net.Send(self)
end

function NetGetdrawcardInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetdrawcardInItBase
