local NetStopXingxiaBase = {}

--终止行侠
function NetStopXingxiaBase:Init(callback)
    self.data = {}
	self.data.msgName = "StopXingxia"

	self.callback = callback

    return self
end

function NetStopXingxiaBase:Send()
    Net.Send(self)
end

function NetStopXingxiaBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetStopXingxiaBase
