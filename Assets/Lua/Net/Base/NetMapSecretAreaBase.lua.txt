local NetMapSecretAreaBase = {}

--进入秘境
-- map_id(string): 地图id
function NetMapSecretAreaBase:Init(map_id,callback)
    self.data = {}
	self.data.msgName = "MapSecretArea"

	self.data.map_id = map_id
	self.callback = callback

    return self
end

function NetMapSecretAreaBase:Send()
    Net.Send(self)
end

function NetMapSecretAreaBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetMapSecretAreaBase
