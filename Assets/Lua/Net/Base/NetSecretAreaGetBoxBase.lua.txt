local NetSecretAreaGetBoxBase = {}

--秘境宝箱领取
-- map_id(string): 地图id
function NetSecretAreaGetBoxBase:Init(map_id,callback)
    self.data = {}
	self.data.msgName = "SecretAreaGetBox"

	self.data.map_id = map_id
	self.callback = callback

    return self
end

function NetSecretAreaGetBoxBase:Send()
    Net.Send(self)
end

function NetSecretAreaGetBoxBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetSecretAreaGetBoxBase
