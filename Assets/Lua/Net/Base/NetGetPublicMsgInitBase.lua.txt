local NetGetPublicMsgInitBase = {}

--初始化获取公告信息
function NetGetPublicMsgInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetPublicMsgInit"

	self.callback = callback

    return self
end

function NetGetPublicMsgInitBase:Send()
    Net.Send(self)
end

function NetGetPublicMsgInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetPublicMsgInitBase
