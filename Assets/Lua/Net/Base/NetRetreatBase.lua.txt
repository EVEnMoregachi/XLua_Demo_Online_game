local NetRetreatBase = {}

--闭关
-- heroId(string): 行舟弟子id
function NetRetreatBase:Init(heroId,callback)
    self.data = {}
	self.data.msgName = "Retreat"

	self.data.heroId = heroId
	self.callback = callback

    return self
end

function NetRetreatBase:Send()
    Net.Send(self)
end

function NetRetreatBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetRetreatBase
