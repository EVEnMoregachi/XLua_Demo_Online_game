local NetInitXiuXingInfoBase = {}

--修行信息初始化
function NetInitXiuXingInfoBase:Init(callback)
    self.data = {}
	self.data.msgName = "InitXiuXingInfo"

	self.callback = callback

    return self
end

function NetInitXiuXingInfoBase:Send()
    Net.Send(self)
end

function NetInitXiuXingInfoBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetInitXiuXingInfoBase
