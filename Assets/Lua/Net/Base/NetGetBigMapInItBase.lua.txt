local NetGetBigMapInItBase = {}

--大地图初始化
function NetGetBigMapInItBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetBigMapInIt"

	self.callback = callback

    return self
end

function NetGetBigMapInItBase:Send()
    Net.Send(self)
end

function NetGetBigMapInItBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetBigMapInItBase
