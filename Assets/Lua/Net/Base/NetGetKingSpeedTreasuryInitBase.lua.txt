local NetGetKingSpeedTreasuryInitBase = {}

--闯王宝库的初始化
function NetGetKingSpeedTreasuryInitBase:Init(callback)
    self.data = {}
	self.data.msgName = "GetKingSpeedTreasuryInit"

	self.callback = callback

    return self
end

function NetGetKingSpeedTreasuryInitBase:Send()
    Net.Send(self)
end

function NetGetKingSpeedTreasuryInitBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetKingSpeedTreasuryInitBase
