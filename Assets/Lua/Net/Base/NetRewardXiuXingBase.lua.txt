local NetRewardXiuXingBase = {}

--修行领奖
-- xiuXingType(string): 修行的10种类型
function NetRewardXiuXingBase:Init(xiuXingType,callback)
    self.data = {}
	self.data.msgName = "RewardXiuXing"

	self.data.xiuXingType = xiuXingType
	self.callback = callback

    return self
end

function NetRewardXiuXingBase:Send()
    Net.Send(self)
end

function NetRewardXiuXingBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetRewardXiuXingBase
