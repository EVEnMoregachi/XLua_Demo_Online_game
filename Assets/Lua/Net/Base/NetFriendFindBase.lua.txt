local NetFriendFindBase = {}

--查找好友（门派名或等级）
-- findDetails(string): 查找内容
function NetFriendFindBase:Init(findDetails,callback)
    self.data = {}
	self.data.msgName = "FriendFind"

	self.data.findDetails = findDetails
	self.callback = callback

    return self
end

function NetFriendFindBase:Send()
    Net.Send(self)
end

function NetFriendFindBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetFriendFindBase
