local NetGetMapShopListBase = {}

--获取商品列表
-- person_id(string): 人物id
function NetGetMapShopListBase:Init(person_id,callback)
    self.data = {}
	self.data.msgName = "GetMapShopList"

	self.data.person_id = person_id
	self.callback = callback

    return self
end

function NetGetMapShopListBase:Send()
    Net.Send(self)
end

function NetGetMapShopListBase:OnReceive(v)
	-- body
    self.responseData = v.value.data[self.data.msgName]
    self:OnResult(v)
    if self.callback ~= nil then
        self.callback(self)
    end
end

return NetGetMapShopListBase
