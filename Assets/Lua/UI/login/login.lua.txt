function login.appendFile(content,mold)  
 	if mold == "account" then  --存账号
		login.clientData["account"] = content
	elseif mold == "music" then  --存音乐设置
		login.clientData["music"] =  content
	elseif mold == "sound" then  --音效设置
		login.clientData["sound"] = content
	elseif mold == "isMoveCam" then  --是否跟随
		login.clientData["isMoveCam"] = content
	elseif mold == "showChat" then  --是否显示聊天
		login.clientData["showChat"] = content
	elseif mold == "showPlayerNum" then  --同屏玩家(1不显示 2少量 3中等 4大量)
		login.clientData["showPlayerNum"] = content
	elseif mold == "serverIp" then  --IP
		login.clientData["serverIp"] = content
	end
	 local str = login.clientData["account"].."|"..login.clientData["music"].."|"..login.clientData["sound"].."|"..login.clientData["isMoveCam"].."|"..login.clientData["showChat"].."|"..login.clientData["showPlayerNum"].."|"..login.clientData["serverIp"]    
	  print("储存=",str)
	  Public.ResourceManager.SetPlayerData(str) 
 
end

  function UITable:DoSendRpc(boxRpc,succussCallback,failCallback)
	
	if #boxRpc == 0 then
		succussCallback()
		return
	end

	local rpc = boxRpc[1]
	table.remove(boxRpc,1)

	local rpcCallback = rpc.callback
	rpc.callback = function ()
		-- print("发送请求返回一次")
		if rpcCallback ~= nil then
			rpcCallback(rpc)
		end
		self:DoSendRpc(boxRpc,succussCallback,failCallback)
	end
	rpc.ErrorHandler = function ()
		failCallback()
	end
	-- print("发送请求一次")
	rpc:Send()
end
