function UITable:CrewItemRenderHandelProp(nIndex,tempTransRect,nIndexTrans) 
	local nDataTotal = 0
	local nNum = 0
	if self.PropItemList[nIndexTrans] == nil then
		self.PropItemList[nIndexTrans] = {}
		setmetatable(self.PropItemList[nIndexTrans] ,UIPropRender)
		self.PropItemList[nIndexTrans]:Init(tempTransRect)
	end

	nDataTotal = #self.PropDataList
 	for i=1,self.oneLineNum do
		nNum = nIndex * self.oneLineNum + i
 		if nNum <= nDataTotal then
            self.PropItemList[nIndexTrans].PropInfo[i]:SetData(self.PropDataList[nNum])
 			self.PropItemList[nIndexTrans].PropInfo[i].transform.gameObject:SetActive(true)
		else
			self.PropItemList[nIndexTrans].PropInfo[i].transform.gameObject:SetActive(false)
		end
	end
end

--刷新背包列表 
function UITable:RefreshCrewList()   
    self.DetailPage:SetActive(false)  
    -- print("刷新背包列表sssssssssssssssssssssssssssssssssss")
    for i = 1, #self.saveList do
        self.saveList[i]:SetActive(true)
     end 
    self.equipChipListObj:SetActive(true)
    self.wuGongChipListObj:SetActive(true) 
	self.PropDataList = GameMainData.PropModelData.PropData  -- 道具主数据
	local nTotal = #self.PropDataList
	local nLine = nTotal/4
    self.propList:SetCount(math.ceil(nLine)) 
 
    
    self.MaterialsDataList = GameMainData.PropModelData.MaterData   -- 道具主数据 材料
	local nTotal = #self.MaterialsDataList
	local nLine = nTotal/4
    self.materList:SetCount(math.ceil(nLine)) 

    if nowPageId == "" then --页面刷新
        self:PageCut(1)
      else
        self:PageCut(nowPageId) 
      end 
 
end  
 
function UITable:InitData()   
    self:heroMsg() --初始化弟子信息
    self:RefreshCrewList()
end

 function UITable:PageCut(pageId) 
    nowPageId = pageId
    self.equipChipListObj:SetActive(false)  --pageId 5
    self.wuGongChipListObj:SetActive(false)  --pageId 6
    for i = 1, #self.saveButton do
        self.saveButton[i]:GetComponent("Image").sprite = self.huibuttondi  
        -- self.saveButton[i]:GetComponent("RectTransform").anchoredPosition3D = CS.UnityEngine.Vector3( 862,self.saveButton[i]:GetComponent("RectTransform").anchoredPosition3D.y,0)
        self.saveList[i]:SetActive(false)
    end

    self.equipB2Obj:SetActive(false)  
    self.equipChipBObj:SetActive(false)  
    self.wugongB2Obj:SetActive(false)  
    self.wuGongChipBObj:SetActive(false) 
    if pageId <=5 then
        self.saveList[pageId]:SetActive(true)  
        self.saveButton[pageId]:GetComponent("Image").sprite = self.huangbuttondi  
        -- self.saveButton[pageId]:GetComponent("RectTransform").anchoredPosition3D = CS.UnityEngine.Vector3(910,self.saveButton[pageId]:GetComponent("RectTransform").anchoredPosition3D.y,0)
   end

        self.equipChipBObj:SetActive(true)

 --弹出道具详情 
function backpack.propDetail(propId) 
    -- print("弹出道具详情",propId) 
    -- print("道具数量",GameMainData.PropModelData:QueryPropNum(propId))
    for i = 1, #save_button do
        CS.UnityEngine.GameObject.Destroy(save_button[i])  -- 清理创建的预制体
    end
    save_button = {} 
    local propMsg = ConfigPropJson.getVarByCustom("prop_id",propId) 
    if propMsg == nil then
        return 
    end
    UITable.det_picture.sprite = Public.ResMeg.GetSpritePath(propId,"backpack") 
    UITable.det_name.text = propMsg.prop_name  
    UITable.det_type.text = UITable:BackPropType(propMsg.prop_type)
    UITable.det_info.text = propMsg.prop_info
    UITable.get_path.text =  propMsg.get_way
    UITable.det_wearer.text=""
    if UITable:BackButtonName(propMsg.prop_btname) ~= ""  then --  按钮对应的功能不同
        local button  = UI.Instantiate(UITable.det_buttonPrefab,UITable.btn_Content,propId)   
        button:GetComponent("Button").onClick:AddListener(function() backpack.UseProps(propId)   end)  
        local buttonname = button.transform:Find("name").gameObject:GetComponent("Text")  --按钮名字 
        buttonname.text = UITable:BackButtonName(propMsg.prop_btname) 
        save_button[1] =  button
        UITable.det_button:SetActive(true)  
    end 
    UITable.DetailPage:SetActive(true)  
end  

--返回按钮Text

	elseif btnType=="2" then
	   return "打开"
	elseif btnType=="3" then

	elseif btnType=="4" then
	   return "传功"		
	elseif btnType=="5" then
	   return "培养"		
	elseif btnType=="6" then
       return "合成" 
    elseif btnType=="7" then
        return "查看" 
	else 
	   return ""		
	end	   
end

 --使用道具
 function backpack.UseProps(prop_Id) 

    print("使用道具id=",prop_Id) 
   local propNum = GameMainData.PropModelData:QueryPropNum(prop_Id)
     print("数量=",propNum)   
   if   propNum <=0 then  --道具数量不足 
    Public.warning("道具数量不足") 
        return 
   end
   usePropId = prop_Id
   local switch = { 
    ["1002"] = function()    --金钥匙
        local propNum2 = GameMainData.PropModelData:QueryPropNum("1003") 
        if propNum2>0 then 
            UITable:UsrPropPage(prop_Id,propNum)  
            if propNum - propNum2 >= 0 then
                canMax = propNum2
                else
                canMax = propNum
            end
            nowNum = 1
        else
            Public.warning("需要金箱子才能开启") 
        end 
    end,
    ["1003"] = function()    --金箱子 
        local propNum2 = GameMainData.PropModelData:QueryPropNum("1002") 
        if propNum2>0 then 
            UITable:UsrPropPage(prop_Id,propNum)  
            if propNum - propNum2 >= 0 then
                canMax = propNum2
                else
                canMax = propNum
            end
            nowNum = 1
