local Render = UI.CreateRenderTable()

function Render:SetUICompent(child) 
 	if child.name == "pic" then   
		 self.picB = child:GetComponent("Button")  
		 self.picB.onClick:RemoveAllListeners()   
		 self.picB.onClick:AddListener(function() self:doSome() end) 
		 self.picImage = child:GetComponent("Image") 
	 elseif child.name == "num" then  
		self.num = child:GetComponent("Text")  
 	end
end

function Render:SetData(v) 
 	self.ScrollData = v 
	self:RefreshScrollItem()
end

function Render:RefreshScrollItem() 
	-- print("self.ScrollData.scrollId =",self.ScrollData.scrollId ) 
 	self.picImage.sprite = Public.ResMeg.GetSpritePath(self.ScrollData.scrollId,"scroll")  
	self.num.text = self.ScrollData.scrollNum
end 

function Render:doSome()   
	backpack.ScrollDetail(self.ScrollData.scrollId) 
end 

return Render
