function ConfigOpenPageInfo.getVarByCustom(name, value)
    for key, var in pairs(configTable) do
        if var[name] == value then
            return var
        end
    end
    return nil
end

function ConfigOpenPageInfo.getVarById(id)
    for key, var in pairs(configTable) do
        if tonumber(var.id) == tonumber(id) then
            return var
        end
    end
    return nil
end

function ConfigOpenPageInfo.getVarByIndex(index)
    return configTable[tonumber(index)]
end

function ConfigOpenPageInfo.getSize()
    return #configTable
end
function ConfigOpenPageInfo.getThisTable ()
  return configTable
end

function  ConfigOpenPageInfo.IfShowMoudle( name ,playerlevel )
-- print("name" ,name)
	local temp = ConfigOpenPageInfo.getVarByCustom("pageName" , name)
	if temp ~= nil then 
		if temp.showLevel <= playerlevel  then
			return true
		else
		    return false
		end 

	end 
	return false
end 

function  ConfigOpenPageInfo.unLockNewMoudle(name,playerlevel  ) 
    local temp = ConfigOpenPageInfo.getVarByCustom("pageName" , name)
   	if temp ~= nil then 
        if temp.showLevel >playerlevel-1  and temp.showLevel == playerlevel   then
             return temp.remakr
		 else 
			 return ""
        end 
    end 
	return ""
end 

return ConfigOpenPageInfo
